```typescript
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
```

**Pattern replacement:**
```tsx
// FROM:
<select value={filter} onChange={(e) => setFilter(e.target.value)} className="...">
  <option value="">Todos</option>
  <option value="foo">Foo</option>
</select>

// TO:
<Select value={filter} onValueChange={setFilter}>
  <SelectTrigger className="w-[180px] h-9">
    <SelectValue placeholder="Todos" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="all">Todos</SelectItem>
    <SelectItem value="foo">Foo</SelectItem>
  </SelectContent>
</Select>
```

**Important notes:**
- `onValueChange` replaces `onChange={(e) => setFilter(e.target.value)}` — shadcn Select passes value directly
- Empty string `""` is not a valid value for SelectItem. Use `"all"` or `"todos"` instead, and update the filter logic to treat `"all"` as no filter
- Keep the same className width but use SelectTrigger's built-in styling
- SuportePage has 2 selects (tipo filter + form tipo) — replace both
- Each file should be read first to understand the exact select structure before replacing

**UX-02 — Replace 3 window.confirm() calls:**

For each file, replace `window.confirm()` with a state-driven `<ConfirmDialog>`.

**Pattern:**
```tsx
// Add state:
const [confirmDelete, setConfirmDelete] = useState<string | null>(null);

// Replace: if (window.confirm('Tem certeza?')) { handleDelete(id); }
// With: setConfirmDelete(id);

// Add dialog at end of JSX:
<ConfirmDialog
  open={!!confirmDelete}
  onOpenChange={(open) => !open && setConfirmDelete(null)}
  title="Confirmar exclusao"
  description="Tem certeza que deseja excluir? Esta acao nao pode ser desfeita."
  onConfirm={() => { handleDelete(confirmDelete!); setConfirmDelete(null); }}
/>
```
