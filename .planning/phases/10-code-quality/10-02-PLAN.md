**QUAL-06 — Migrate 7 simpler hooks.**

For each hook, apply this transformation pattern:

**FROM (manual state management):**
```typescript
const [data, setData] = useState<T[]>([]);
const [loading, setLoading] = useState(true);
const [error, setError] = useState<string | null>(null);

useEffect(() => {
  const fetchData = async () => {
    setLoading(true);
    const { data, error } = await supabase.from('table').select('*');
    if (error) { setError(error.message); } else { setData(data); }
    setLoading(false);
  };
  fetchData();
}, [deps]);

return { data, loading, error };
```

**TO (React Query):**
```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

const { data = [], isLoading: loading, isError, error } = useQuery({
  queryKey: ['table-name', ...deps],
  queryFn: async () => {
    const { data, error } = await supabase.from('table').select('*');
    if (error) throw error;
    return data;
  },
});

// For mutations (create/update/delete), use useMutation:
const queryClient = useQueryClient();
const createMutation = useMutation({
  mutationFn: async (newItem: T) => {
    const { data, error } = await supabase.from('table').insert(newItem).select().single();
    if (error) throw error;
    return data;
  },
  onSuccess: () => {
    queryClient.invalidateQueries({ queryKey: ['table-name'] });
    toast({ title: 'Sucesso' });
  },
  onError: (err) => {
    toast({ title: 'Erro', description: err.message, variant: 'destructive' });
  },
});

return { data, loading, createMutation, ... };
```

**Important migration rules:**
1. Preserve the SAME return interface so consumers don't break
2. If the hook returns `{ loading }`, alias `isLoading` as `loading`
3. If the hook returns `{ error: string }`, extract `error.message` from the Error object
4. Keep any `useEffect` for realtime subscriptions — only migrate the initial fetch and mutations
5. Keep the same queryKey namespace consistent with existing hooks (check for cache invalidation patterns)
